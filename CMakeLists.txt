cmake_minimum_required(VERSION 3.16)
project(fakepanic VERSION 1.0.1 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
add_executable(fakepanic-renderer src/main.cpp)
target_compile_features(fakepanic-renderer PRIVATE cxx_std_17)
target_compile_options(fakepanic-renderer PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(fakepanic-renderer PRIVATE FAKEPANIC_VERSION="${PROJECT_VERSION}")
target_link_libraries(fakepanic-renderer PRIVATE PkgConfig::SDL2 PkgConfig::SDL2_TTF Threads::Threads)
install(TARGETS fakepanic-renderer DESTINATION ${CMAKE_INSTALL_DATADIR}/fakepanic)
install(PROGRAMS scripts/fakepanic DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/start-game.sh scripts/input-bridge.py DESTINATION ${CMAKE_INSTALL_DATADIR}/fakepanic/scripts)
install(FILES examples/bad-apple.mp4 examples/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/fakepanic/examples)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fakepanic)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(fakepanic-visual-probe SHARED tests/visual_probe.cpp)
  target_link_libraries(fakepanic-visual-probe PRIVATE PkgConfig::SDL2 ${CMAKE_DL_LIBS})
  target_compile_features(fakepanic-visual-probe PRIVATE cxx_std_17)
  add_test(NAME visual COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/visual.py $<TARGET_FILE:fakepanic-renderer> $<TARGET_FILE:fakepanic-visual-probe>)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:fakepanic-renderer>)
endif()
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Fake Panic contributors <79757617+George-Salt@users.noreply.github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Fake Panic: video in a simulated Linux panic screen")
set(CPACK_PACKAGE_NAME fakepanic)
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "ffmpeg, fonts-dejavu-core, python3")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SUGGESTS "gamescope, pipewire, gstreamer1.0-pipewire, gstreamer1.0-tools, gstreamer1.0-plugins-base, jq, libx11-6, libxtst6")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_FILE_NAME "fakepanic-${PROJECT_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
